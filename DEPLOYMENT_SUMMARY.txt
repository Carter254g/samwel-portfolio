  POST   /api/admin/login              ← Authenticate admin users
  GET    /api/admin/portfolio          ← Fetch all portfolio images
  POST   /api/admin/portfolio          ← Add new portfolio image
  PUT    /api/admin/portfolio          ← Update portfolio image
  DELETE /api/admin/portfolio          ← Delete portfolio image

  GET    /api/admin/services           ← Fetch all services
  POST   /api/admin/services           ← Add new service
  PUT    /api/admin/services           ← Update service
  DELETE /api/admin/services           ← Delete service

  GET    /api/admin/testimonials       ← Fetch all testimonials
  POST   /api/admin/testimonials       ← Add new testimonial
  PUT    /api/admin/testimonials       ← Update testimonial
  DELETE /api/admin/testimonials       ← Delete testimonial

  GET    /api/admin/contacts           ← View contact submissions
  PUT    /api/admin/contacts           ← Mark read/archived
  DELETE /api/admin/contacts           ← Delete contact

  POST   /api/contact                  ← Public contact form submission
